# Phase 5: Dashboard & Reports — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give the employer a dashboard that shows today's crossings live, per-employee out-of-work time, and the video evidence — plus daily CSV/XLSX and monthly PDF reports they can hand to an accountant or use to settle a dispute.

**Architecture:** All reporting logic lives in a pure `reports/` package that reads `sessions`/`events` and writes CSV/XLSX/PDF; the service only exposes a download endpoint. The Electron UI gains three panels (live view with zone overlay, records, reports) plus a clip player that tolerates in-progress encoding. Everything is testable headlessly with a seeded temp DB.

**Tech Stack:** Electron (existing), openpyxl (XLSX), reportlab (PDF), Python csv (stdlib). No new service dependencies.

**Phases 1–4 must be complete.** Interfaces relied on here (all listed in the Phase 4 handoff):
- `GET /events?since=`, `GET /sessions?date=`, `GET /clips?date=`
- `GET/POST /zones`, `GET /employees`, `POST /employees/{id}/deactivate`
- `GET /resolver/status`, `GET /pipeline/status`, `GET /health`, `GET /health/rss`
- `POST /retention/purge`
- Phase 3 note: `ClipRecorder.capture()` returns *before* the file finishes writing (up to `post_seconds`). The clip player must tolerate a missing/partial file.

## Global Constraints

- Target: Windows 11, 16 GB DDR4, 6th-gen Intel, no dGPU. **One `BrowserWindow`.** The live view is a single MJPEG `<img>`; no per-frame IPC, no second renderer process.
- Reports are generated by the **service** (it owns the filesystem and the DB), then downloaded by the UI. The renderer never reads the DB.
- Report output directory: `<data_root>\reports\`. Cleaned by the same retention job as clips, with its own setting.
- All monetary-adjacent numbers (minutes) are computed in `reports/`, never in SQL aggregates that can silently drop soft-deleted rows. Soft-deleted sessions are excluded from reports.
- Auth: reports, settings, and record edits require the local password. Implemented in Phase 5 as a UI gate plus a service-side check on mutating endpoints.

```python
# reports/query.py
"""The only module that reads sessions/events for reporting.
Every query excludes soft-deleted rows — this is the single enforcement point."""
import sqlite3
from datetime import datetime
from typing import Optional
from .schema import SessionRow


def sessions_for_date(conn: sqlite3.Connection, date: str) -> list[SessionRow]:
    rows = conn.execute(
        """SELECT s.id, s.employee_id, e.name AS employee_name, s.out_at, s.in_at,
                  s.duration_minutes, s.inferred_return, s.date,
                  eo.clip_path AS out_clip, ei.clip_path AS in_clip
             FROM sessions s
             JOIN employees e ON e.id = s.employee_id
             LEFT JOIN events eo ON eo.id = s.out_event_id
             LEFT JOIN events ei ON ei.id = s.in_event_id
            WHERE s.date = ? AND s.deleted_at IS NULL
            ORDER BY s.out_at""", (date,)).fetchall()
    return [_row(r) for r in rows]


def sessions_for_employee(conn, employee_id: int, start: str, end: str) -> list[SessionRow]:
    rows = conn.execute(
        """SELECT s.id, s.employee_id, e.name AS employee_name, s.out_at, s.in_at,
                  s.duration_minutes, s.inferred_return, s.date,
                  eo.clip_path AS out_clip, ei.clip_path AS in_clip
             FROM sessions s
             JOIN employees e ON e.id = s.employee_id
             LEFT JOIN events eo ON eo.id = s.out_event_id
             LEFT JOIN events ei ON ei.id = s.in_event_id
            WHERE s.employee_id = ? AND s.date BETWEEN ? AND ?
                  AND s.deleted_at IS NULL
            ORDER BY s.out_at""", (employee_id, start, end)).fetchall()
    return [_row(r) for r in rows]


def employee_daily_totals(conn: sqlite3.Connection, date: str) -> list[dict]:
    rows = conn.execute(
        """SELECT employee_id, COUNT(*) AS trips,
                  COALESCE(SUM(duration_minutes), 0) AS minutes
             FROM sessions
            WHERE date = ? AND deleted_at IS NULL
            GROUP BY employee_id ORDER BY employee_id""", (date,)).fetchall()
    return [dict(r) for r in rows]


def events_since(conn: sqlite3.Connection, iso: str) -> list[sqlite3.Row]:
    return conn.execute(
        """SELECT ev.*, e.name AS employee_name FROM events ev
             JOIN employees e ON e.id = ev.employee_id
            WHERE ev.occurred_at >= ? ORDER BY ev.id""", (iso,)).fetchall()


def _row(r: sqlite3.Row) -> SessionRow:
    return SessionRow(
        id=r["id"], employee_id=r["employee_id"], employee_name=r["employee_name"],
        out_at=r["out_at"], in_at=r["in_at"],
        duration_minutes=r["duration_minutes"],
        inferred_return=bool(r["inferred_return"]), date=r["date"],
        out_clip=r["out_clip"], in_clip=r["in_clip"])
```

```python
# reports/daily.py
"""Daily CSV + XLSX. Contains no SQL — it formats rows from reports/query.py."""
import csv
from pathlib import Path
import sqlite3
from .query import sessions_for_date, employee_daily_totals

HEADERS = ["employee_id", "employee_name", "date", "out_at", "in_at",
           "duration_minutes", "inferred_return", "out_clip", "in_clip"]


def write_daily_csv(conn: sqlite3.Connection, date: str, out_dir: Path) -> Path:
    out_dir = Path(out_dir); out_dir.mkdir(parents=True, exist_ok=True)
    path = out_dir / f"attendance_{date}.csv"
    rows = sessions_for_date(conn, date)
    with path.open("w", newline="", encoding="utf-8") as fh:
        w = csv.DictWriter(fh, fieldnames=HEADERS)
        w.writeheader()
        for r in rows:
            w.writerow({h: getattr(r, h) for h in HEADERS})
    return path


def write_daily_xlsx(conn: sqlite3.Connection, date: str, out_dir: Path) -> Path:
    from openpyxl import Workbook
    out_dir = Path(out_dir); out_dir.mkdir(parents=True, exist_ok=True)
    path = out_dir / f"attendance_{date}.xlsx"
    rows = sessions_for_date(conn, date)
    totals = employee_daily_totals(conn, date)

    wb = Workbook()
    ws = wb.active; ws.title = "Trips"
    ws.append(HEADERS)
    for r in rows:
        ws.append([getattr(r, h) for h in HEADERS])
    for col, width in zip("ABCDEFGHI", (12, 22, 12, 21, 21, 17, 16, 34, 34)):
        ws.column_dimensions[col].width = width

    s = wb.create_sheet("Summary")
    s.append(["employee_id", "trips", "total_minutes_out"])
    for t in totals:
        s.append([t["employee_id"], t["trips"], round(t["minutes"], 1)])
    s.append([])
    s.append(["TOTAL", sum(t["trips"] for t in totals),
              round(sum(t["minutes"] for t in totals), 1)])
    wb.save(path)
    return path
```

```python
# reports/monthly.py
"""Monthly per-employee PDF. No SQL here — formatting only."""
import calendar
from pathlib import Path
import sqlite3
from .query import sessions_for_employee
from .roster import list_employees


def write_monthly_pdf(conn: sqlite3.Connection, year_month: str, out_dir: Path) -> Path:
    from reportlab.lib.pagesizes import A4
    from reportlab.lib.units import mm
    from reportlab.pdfgen import canvas

    year, month = (int(x) for x in year_month.split("-"))
    last_day = calendar.monthrange(year, month)[1]
    start, end = f"{year_month}-01", f"{year_month}-{last_day:02d}"

    out_dir = Path(out_dir); out_dir.mkdir(parents=True, exist_ok=True)
    path = out_dir / f"summary_{year_month}.pdf"
    c = canvas.Canvas(str(path), pagesize=A4)
    w, h = A4
    y = h - 20 * mm

    def line(txt, dy=6 * mm, bold=False):
        nonlocal y
        if y < 20 * mm:
            c.showPage(); y = h - 20 * mm
        c.setFont("Helvetica-Bold" if bold else "Helvetica", 11 if bold else 9)
        c.drawString(20 * mm, y, txt); y -= dy

    line(f"Employee out-of-work summary — {year_month}", 10 * mm, bold=True)
    line("")
    employees = list_employees(conn)
    grand = 0.0
    for emp in employees:
        rows = sessions_for_employee(conn, emp["id"], start, end)
        total = sum(r.duration_minutes or 0 for r in rows)
        grand += total
        line(f"{emp['name']}  ({emp['employee_code'] or '-'})", 5 * mm, bold=True)
        line(f"  trips: {len(rows)}   total out: {total:.1f} minutes")
        for r in rows:
            flag = "inferred" if r.inferred_return else ""
            line(f"    {r.out_at[11:16]} – {(r.in_at or '??')[11:16]}  "
                 f"{r.duration_minutes or 0:.0f} min {flag}")
        line("")
    line(f"Roster total: {grand:.1f} minutes", 8 * mm, bold=True)
    c.save()
    return path
```

```python
# service/password.py
"""Local password gate. Never stores or compares plaintext."""
import hashlib, hmac, os, sqlite3

ITERATIONS = 240_000
KEY = "password_hash"


class PasswordGate:
    def __init__(self, conn: sqlite3.Connection):
        self.conn = conn

    def _read(self) -> str | None:
        row = self.conn.execute("SELECT value FROM settings WHERE key=?", (KEY,)).fetchone()
        return row[0] if row else None

    def is_set(self) -> bool:
        return self._read() is not None

    def set(self, plaintext: str) -> None:
        salt = os.urandom(16)
        dk = hashlib.pbkdf2_hmac("sha256", plaintext.encode(), salt, ITERATIONS)
        self.conn.execute(
            "INSERT INTO settings (key,value) VALUES (?,?) "
            "ON CONFLICT(key) DO UPDATE SET value=excluded.value",
            (KEY, f"pbkdf2_sha256${ITERATIONS}${salt.hex()}${dk.hex()}"))
        self.conn.commit()

    def verify(self, plaintext: str) -> bool:
        stored = self._read()
        if not stored:
            return False
        try:
            _, iters, salt_hex, hash_hex = stored.split("$")
            dk = hashlib.pbkdf2_hmac("sha256", plaintext.encode(),
                                     bytes.fromhex(salt_hex), int(iters))
        except ValueError:
            return False
        return hmac.compare_digest(dk.hex(), hash_hex)

    def require(self, plaintext: str) -> None:
        if not self.verify(plaintext):
            raise PermissionError("password required or incorrect")
```

```python
# service/mjpeg.py
"""Multipart MJPEG at low fps. Deliberately cheap: the overlay carries the meaning."""
import queue, threading, time
from pathlib import Path
import cv2
import numpy as np
from .zone import get_active_zone


class MjpegServer:
    def __init__(self, max_fps: float = 6.0):
        self.max_fps = max_fps
        self._frame: bytes | None = None
        self._lock = threading.Lock()
        self._subs: list[queue.Queue] = []
        self._stop = threading.Event()

    def publish(self, image: np.ndarray) -> None:
        ok, jpg = cv2.imencode(".jpg", image, [int(cv2.IMWRITE_JPEG_QUALITY), 70])
        if not ok:
            return
        with self._lock:
            self._frame = jpg.tobytes()
            subs = list(self._subs)
        for q in subs:
            try: q.put_nowait(self._frame)
            except queue.Full: pass

    def subscribe(self) -> queue.Queue:
        q: queue.Queue = queue.Queue(maxsize=2)
        with self._lock:
            self._subs.append(q)
        return q

    def unsubscribe(self, q: queue.Queue) -> None:
        with self._lock:
            if q in self._subs:
                self._subs.remove(q)

    def stop(self) -> None:
        self._stop.set()
```

```python
# service/cap.py
"""Defensive clip helpers. Clips may legitimately be mid-write, unreadable, or missing."""
import json
from pathlib import Path
from typing import Optional


def read_timespan_json(path: Path) -> Optional[dict]:
    try:
        data = json.loads(Path(path).read_text(encoding="utf-8"))
    except Exception:
        return None
    if not isinstance(data, dict):
        return None
    return data


def probe_clip(path: Path) -> dict:
    p = Path(path)
    if not p.exists():
        return {"exists": False, "frames": 0, "opened": False, "ready": False}
    try:
        import cv2
        cap = cv2.VideoCapture(str(p))
        opened = cap.isOpened()
        frames = int(cap.get(7)) if opened else 0
        cap.release()
    except Exception:
        return {"exists": True, "frames": 0, "opened": False, "ready": False}
    return {"exists": True, "frames": frames, "opened": opened, "ready": frames > 0}


def make_probe_payload() -> dict:
    return {"version": 1, "note": "metadata written alongside each clip for diagnostics"}
```
